> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a connected "오늘 시장 분위기" dashboard briefing that links market regime to focus/watch sectors and representative tickers.

**Architecture:** Keep the output JSON contract unchanged and derive sector mood insights in the frontend from `day.market_regime`, `day.macro_context`, and `day.tickers`. ```ts
import type { MacroContext, MacroEvent, MarketRegimeData, TickerAnalysisData } from '../types'

export type SectorMoodClassification = 'focus' | 'neutral' | 'watch'
export type SectorFitLabel = '정합 높음' | '중립' | '정합 낮음' | '확인 필요'

export interface SectorTickerRef {
  ticker: string
  name: string
  change: number | null
  action?: string
  conviction?: number
}

export interface SectorMoodInsight {
  sector: string
  label: string
  tickerCount: number
  averageDailyChange: number | null
  positiveTickerRatio: number | null
  topGainer: SectorTickerRef | null
  topLoser: SectorTickerRef | null
  representativeTickers: SectorTickerRef[]
  regimeFitLabel: SectorFitLabel
  classification: SectorMoodClassification
  score: number
  priceFlowScore: number
  regimeFitScore: number
  tickerSignalScore: number
  eventExposureScore: number
  rationale: string
  macroEvidence: string[]
}

export interface SectorMoodResult {
  insights: SectorMoodInsight[]
  focus: SectorMoodInsight[]
  watch: SectorMoodInsight[]
  neutral: SectorMoodInsight[]
  hasSectorData: boolean
  emptyReason?: string
}

interface DeriveSectorMoodInput {
  tickers: TickerAnalysisData[]
  marketRegime?: MarketRegimeData | null
  macroContext?: MacroContext | null
}

interface SectorBucket {
  sector: string
  tickers: SectorTickerRef[]
  changes: number[]
  factorScores: number[]
  tickerSignals: number[]
}

const SECTOR_LABELS: Record<string, string> = {
  Technology: '기술',
  Semiconductors: '반도체',
  Healthcare: '헬스케어',
  Financials: '금융',
  Energy: '에너지',
  'Consumer Discretionary': '경기소비재',
  'Consumer Staples': '필수소비재',
  Industrials: '산업재',
  'Communication Services': '커뮤니케이션',
  Utilities: '유틸리티',
  'Real Estate': '부동산',
  Materials: '소재',
  Other: '기타',
}

const FACTOR_KEYS = ['macro_regime', 'regime_adjustment', 'macro_event']
const FOCUS_LIMIT = 3
const WATCH_LIMIT = 3

export function deriveSectorMoodInsights({
  tickers,
  marketRegime,
  macroContext,
}: DeriveSectorMoodInput): SectorMoodResult {
  const buckets = buildSectorBuckets(tickers)

  if (buckets.length === 0) {
    return {
      insights: [],
      focus: [],
      watch: [],
      neutral: [],
      hasSectorData: false,
      emptyReason: '섹터 데이터 부족',
    }
  }

  const rawInsights = buckets.map((bucket) => buildRawInsight(bucket, marketRegime, macroContext))
  const sortedByScore = [...rawInsights].sort((left, right) => right.score - left.score)
  const focusKeys = new Set(
    sortedByScore
      .filter((insight) => insight.score >= 0.12 && insight.regimeFitScore > -0.2 && insight.priceFlowScore > -0.35)
      .slice(0, FOCUS_LIMIT)
      .map((insight) => insight.sector),
  )
  const watchKeys = new Set(
    [...sortedByScore]
      .reverse()
      .filter(
        (insight) =>
          !focusKeys.has(insight.sector) &&
          (insight.score <= -0.08 ||
            insight.regimeFitScore <= -0.3 ||
            insight.priceFlowScore <= -0.35 ||
            insight.eventExposureScore <= -0.35),
      )
      .slice(0, WATCH_LIMIT)
      .map((insight) => insight.sector),
  )

  const insights = sortedByScore.map((insight) => {
    const classification: SectorMoodClassification = focusKeys.has(insight.sector)
      ? 'focus'
      : watchKeys.has(insight.sector)
        ? 'watch'
        : 'neutral'
    return {
      ...insight,
      classification,
      regimeFitLabel: getFitLabel(classification, insight.regimeFitScore),
      rationale: buildRationale(insight, classification, marketRegime),
    }
  })

  return {
    insights,
    focus: insights.filter((insight) => insight.classification === 'focus'),
    watch: insights.filter((insight) => insight.classification === 'watch'),
    neutral: insights.filter((insight) => insight.classification === 'neutral'),
    hasSectorData: true,
  }
}

export function buildMarketMoodSummary(
  regime: MarketRegimeData | null | undefined,
  sectorMood: SectorMoodResult,
): string {
  const regimeLabel = regime?.regime ?? '시장 분위기 정보 없음'
  if (!sectorMood.hasSectorData || (sectorMood.focus.length === 0 && sectorMood.watch.length === 0)) {
    return `${regimeLabel} · 매크로와 섹터 흐름`
  }
  return `${regimeLabel} · 주목 ${sectorMood.focus.length}개 / 주의 ${sectorMood.watch.length}개 · 매크로와 섹터 흐름`
}

export function getSectorLabel(sector: string): string {
  return SECTOR_LABELS[sector] ?? sector
}

export function formatSectorPercent(value: number | null): string {
  if (value === null || !Number.isFinite(value)) return 'N/A'
  return `${value >= 0 ? '+' : ''}${value.toFixed(2)}%`
}

function buildSectorBuckets(tickers: TickerAnalysisData[]): SectorBucket[] {
  const bucketMap = new Map<string, SectorBucket>()

  for (const ticker of tickers) {
    const sector = ticker.data_snapshot['Sector']?.trim() || 'Other'
    const change = parseOptionalPercent(ticker.data_snapshot['Daily Change'])
    const current = bucketMap.get(sector) ?? {
      sector,
      tickers: [],
      changes: [],
      factorScores: [],
      tickerSignals: [],
    }

    const ref: SectorTickerRef = {
      ticker: ticker.ticker,
      name: ticker.name,
      change,
      action: ticker.decision?.action,
      conviction: ticker.decision?.conviction,
    }

    current.tickers.push(ref)
    if (change !== null) current.changes.push(change)
    current.factorScores.push(...extractFactorScores(ticker))
    current.tickerSignals.push(scoreTickerSignal(ticker))
    bucketMap.set(sector, current)
  }

  return Array.from(bucketMap.values()).filter((bucket) => bucket.tickers.length > 0)
}

function buildRawInsight(
  bucket: SectorBucket,
  marketRegime?: MarketRegimeData | null,
  macroContext?: MacroContext | null,
): SectorMoodInsight {
  const averageDailyChange = bucket.changes.length > 0 ? average(bucket.changes) : null
  const positiveTickerRatio = bucket.changes.length > 0
    ? bucket.changes.filter((change) => change > 0).length / bucket.changes.length
    : null
  const priceFlowScore = scorePriceFlow(averageDailyChange, positiveTickerRatio)
  const regimeFitScore = bucket.factorScores.length > 0 ? average(bucket.factorScores) : 0
  const tickerSignalScore = bucket.tickerSignals.length > 0 ? average(bucket.tickerSignals) : 0
  const macroEvidence = collectMacroEvidence(bucket.sector, macroContext)
  const eventExposureScore = scoreEventExposure(macroEvidence, macroContext)
  const score = roundScore(
    priceFlowScore * 0.35 +
      regimeFitScore * 0.35 +
      tickerSignalScore * 0.2 +
      eventExposureScore * 0.1,
  )
  const sortedByChange = bucket.tickers
    .filter((ticker) => ticker.change !== null)
    .sort((left, right) => (right.change ?? 0) - (left.change ?? 0))

  return {
    sector: bucket.sector,
    label: getSectorLabel(bucket.sector),
    tickerCount: bucket.tickers.length,
    averageDailyChange,
    positiveTickerRatio,
    topGainer: sortedByChange[0] && (sortedByChange[0].change ?? 0) > 0 ? sortedByChange[0] : null,
    topLoser:
      sortedByChange[sortedByChange.length - 1] && (sortedByChange[sortedByChange.length - 1].change ?? 0) < 0
        ? sortedByChange[sortedByChange.length - 1]
        : null,
    representativeTickers: buildRepresentativeTickers(bucket.tickers),
    regimeFitLabel: getFitLabel('neutral', regimeFitScore),
    classification: 'neutral',
    score,
    priceFlowScore,
    regimeFitScore,
    tickerSignalScore,
    eventExposureScore,
    rationale: buildRationale(
      {
        sector: bucket.sector,
        label: getSectorLabel(bucket.sector),
        priceFlowScore,
        regimeFitScore,
        eventExposureScore,
        averageDailyChange,
      },
      'neutral',
      marketRegime,
    ),
    macroEvidence,
  }
}

function extractFactorScores(ticker: TickerAnalysisData): number[] {
  const factors = ticker.decision?.factors ?? {}
  return FACTOR_KEYS
    .map((key) => factors[key])
    .filter((value): value is number => typeof value === 'number' && Number.isFinite(value))
    .map(normalizeFactorScore)
}

function scoreTickerSignal(ticker: TickerAnalysisData): number {
  const action = ticker.decision?.action
  const actionScore = action === 'buy' ? 1 : action === 'avoid' ? -1 : 0
  const conviction = ticker.decision?.conviction
  const convictionScore = typeof conviction === 'number' ? clamp((conviction - 50) / 50, -1, 1) : 0
  return roundScore(actionScore * 0.65 + convictionScore * 0.35)
}

function scorePriceFlow(averageDailyChange: number | null, positiveTickerRatio: number | null): number {
  const changeScore = averageDailyChange === null ? 0 : clamp(averageDailyChange / 3, -1, 1)
  const breadthScore = positiveTickerRatio === null ? 0 : clamp((positiveTickerRatio - 0.5) * 2, -1, 1)
  return roundScore(changeScore * 0.7 + breadthScore * 0.3)
}

function collectMacroEvidence(sector: string, macroContext?: MacroContext | null): string[] {
  const sectorLabel = getSectorLabel(sector)
  const events = [...(macroContext?.macro_events ?? []), ...(macroContext?.upcoming_macro_events ?? [])]
  return events
    .filter((event) => macroEventMatchesSector(event, sector, sectorLabel))
    .map((event) => event.summary_ko || event.market_bias || `${event.label} 영향 확인`)
    .filter((text) => text.trim().length > 0)
    .slice(0, 3)
}

function macroEventMatchesSector(event: MacroEvent, sector: string, sectorLabel: string): boolean {
  const tags = [
    ...(event.affected_sectors ?? []),
    ...(event.affected_industries ?? []),
    ...normalizeTags(event.sensitivity_tags),
  ]
  const normalizedSector = normalizeText(sector)
  const normalizedLabel = normalizeText(sectorLabel)
  return tags.some((tag) => {
    const normalizedTag = normalizeText(tag)
    return (
      normalizedTag.includes(normalizedSector) ||
      normalizedSector.includes(normalizedTag) ||
      normalizedTag.includes(normalizedLabel) ||
      normalizedLabel.includes(normalizedTag)
    )
  })
}

function scoreEventExposure(evidence: string[], macroContext?: MacroContext | null): number {
  if (evidence.length === 0) return 0
  const joined = evidence.join(' ').toLowerCase()
  if (/(부담|압박|위험|둔화|hawkish|negative|headwind|risk|volatility)/i.test(joined)) return -0.6
  if (/(우호|완화|강세|dovish|positive|tailwind|support)/i.test(joined)) return 0.45
  const hasHighSeverity = [...(macroContext?.macro_events ?? []), ...(macroContext?.upcoming_macro_events ?? [])].some(
    (event) => event.severity === 'high' || event.impact === 'high',
  )
  return hasHighSeverity ? -0.2 : 0.1
}

function buildRepresentativeTickers(tickers: SectorTickerRef[]): SectorTickerRef[] {
  return [...tickers]
    .sort((left, right) => {
      const signalDiff = tickerRefSignal(right) - tickerRefSignal(left)
      if (signalDiff !== 0) return signalDiff
      return (right.change ?? Number.NEGATIVE_INFINITY) - (left.change ?? Number.NEGATIVE_INFINITY)
    })
    .slice(0, 4)
}

function tickerRefSignal(ticker: SectorTickerRef): number {
  const actionScore = ticker.action === 'buy' ? 2 : ticker.action === 'avoid' ? -1 : 0
  return actionScore + (ticker.conviction ?? 50) / 100
}

function getFitLabel(classification: SectorMoodClassification, regimeFitScore: number): SectorFitLabel {
  if (classification === 'focus' || regimeFitScore >= 0.25) return '정합 높음'
  if (classification === 'watch' || regimeFitScore <= -0.25) return '정합 낮음'
  if (Math.abs(regimeFitScore) < 0.12) return '확인 필요'
  return '중립'
}

function buildRationale(
  insight: Pick<SectorMoodInsight, 'label' | 'priceFlowScore' | 'regimeFitScore' | 'eventExposureScore' | 'averageDailyChange'>,
  classification: SectorMoodClassification,
  marketRegime?: MarketRegimeData | null,
): string {
  const regimeText = marketRegime?.regime ? `${marketRegime.regime} 분위기` : '현재 시장 분위기'
  if (classification === 'focus') {
    if (insight.priceFlowScore > 0.2 && insight.regimeFitScore >= 0) {
      return `${regimeText}와 가격 흐름이 같은 방향입니다.`
    }
    return `${regimeText}와 종목 신호가 비교적 잘 맞습니다.`
  }
  if (classification === 'watch') {
    if (insight.eventExposureScore < -0.3) {
      return `매크로 이벤트 노출이 있어 변동성과 상대강도를 확인할 섹터입니다.`
    }
    if (insight.regimeFitScore < -0.25) {
      return `${regimeText}와 정합도가 낮아 우선순위를 낮춰 확인합니다.`
    }
    return `가격 흐름이 약해 반등 전까지 상대강도 점검이 필요합니다.`
  }
  if (insight.averageDailyChange === null) {
    return `일일 등락 데이터가 부족해 중립으로 표시합니다.`
  }
  return `흐름과 정합도 신호가 엇갈려 중립으로 확인합니다.`
}

function parseOptionalPercent(value?: string): number | null {
  if (!value || value === 'N/A') return null
  const match = value.replace(/,/g, '').match(/[-+]?\d*\.?\d+/)
  if (!match) return null
  const parsed = Number.parseFloat(match[0])
  return Number.isFinite(parsed) ? parsed : null
}

function normalizeFactorScore(value: number): number {
  if (Math.abs(value) <= 1) return roundScore(clamp(value, -1, 1))
  return roundScore(clamp(value / 2, -1, 1))
}

function normalizeTags(value?: string | string[]): string[] {
  if (!value) return []
  return Array.isArray(value) ? value : value.split(',')
}

function normalizeText(value: string): string {
  return value.trim().toLowerCase()
}

function average(values: number[]): number {
  return values.reduce((sum, value) => sum + value, 0) / values.length
}

function clamp(value: number, min: number, max: number): number {
  return Math.min(max, Math.max(min, value))
}

function roundScore(value: number): number {
  return Math.round(value * 1000) / 1000
}

const REGIME_LABELS: Record<string, string> = {

                {ticker.ticker}

            ))}
          </div>

      </div>

function SectorMoodComparison({ insights }: { insights: SectorMoodInsight[] }) {

  return (

      <h4>섹터 전체 비교</h4>

          <span>등락</span>
          <span>판정</span>

function classificationLabel(classification: SectorMoodInsight['classification']): string {

}

function collectEvidence(result: SectorMoodResult): string[] {
  return Array.from(new Set(result.insights.flatMap((insight) => insight.macroEvidence))).slice(0, 3)
}
